\begin{enumerate}

\item
Let $n$ be a positive integer and let
$a_1,a_2,...,a_k$ $(k >= 2)$ be distinct integers in $\{1,...,n\}$
such that $n$ divides $a_i(a_{i+1}-1)$ for $i = 1,...,k-1$.
Prove that $n$ does not divide $a_k(a_1-1)$.

\item
Let $ABC$ be a triangle with circumcentre $O$.
Points $P$ and $Q$ are interior points of sides $CA$ and $AB$.
Let $K,L,M$ be the midpoints of $BP,CQ,PQ$ respectively.
Let a circle pass through $K,L,M$.
Suppose $PQ$ is tangent to this circle.
Prove that $OP = OQ$.

\item
Suppose that $s_1,s_2,s_3,...$ is a strictly increasing sequence
of positive integers such that the subsequences
$s_1,s_3,s_5,...$ and $s_2,s_4,s_6,...$
are arithmetic progressions.
Prove that $s_1,s_2,s_3,...$ is an arithmetic progression.

\item
Let $ABC$ be a triangle with $AB = AC$.
The angle bisectors of $CAB$ and $ABC$
meet sides $BC$ and $CA$ at $D$ and $E$ respectively.
Let $K$ be the incenter of triangle $ADC$.
Suppose $BEK = 45$ degrees.
Find all possible values of $CAB$.

\item
Determine all functions $f$ from the set of positive integers
to the set of positive integers such that for all positive integers
$a$ and $b$, there exists a non degenerate triangle with sides
$a$, $f(b)$ and $f(a)-1$.

\item
Let $a_1,a_2,...,a_n$ be distinct positive integers and let
$M$ be a set of $n-1$ positive integers containing
$s = a_1 + a_2 + ... + a_n$.
A grasshopper starts at $0$ on the real axis and makes jumps
of lengths $a_1,a_2,...,a_n$ in some order.
Prove that the order can be chosen so that the grasshopper
never lands on any point of $M$.

\end{enumerate}
